// sw.js — simpan di folder yang sama dengan index.html const CACHE = 'kadar-emas'; self.addEventListener('install', (e) => { e.waitUntil(caches.open(CACHE).then((c) => c.addAll(['./']))); self.skipWaiting(); }); self.addEventListener('activate', (e) => { e.waitUntil( caches.keys() .then((keys) => Promise.all(keys.filter((k) => k !== CACHE).map((k) => caches.delete(k)))) .then(() => self.clients.claim()) ); }); // Online: ambil versi terbaru lalu simpan ke cache. Offline: pakai cache. self.addEventListener('fetch', (e) => { if (e.request.method !== 'GET') return; e.respondWith( fetch(e.request) .then((res) => { const salinan = res.clone(); caches.open(CACHE).then((c) => c.put(e.request, salinan)); return res; }) .catch(() => caches.match(e.request, { ignoreSearch: true }).then((r) => r || caches.match('./')) ) ); });
