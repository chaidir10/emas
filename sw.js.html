// sw.js — simpan di folder yang sama dengan index.html const CACHE = 'kadar-emas-v5'; self.addEventListener('install', () => self.skipWaiting()); self.addEventListener('activate', (e) => { e.waitUntil( caches.keys() .then((keys) => Promise.all(keys.filter((k) => k !== CACHE).map((k) => caches.delete(k)))) .then(() => self.clients.claim()) ); }); // Tampilkan dari cache dulu; perbarui cache di latar belakang bila online self.addEventListener('fetch', (e) => { const req = e.request; if (req.method !== 'GET' || new URL(req.url).origin !== location.origin) return; const jaringan = fetch(req).then((res) => { if (res.ok && !res.redirected) { const salinan = res.clone(); caches.open(CACHE).then((c) => c.put(req, salinan)); } return res; }); e.waitUntil(jaringan.catch(() => {})); e.respondWith( caches.open(CACHE).then(async (c) => { let cached = await c.match(req, { ignoreSearch: true }); if (!cached && req.mode === 'navigate') { const kunci = await c.keys(); // pakai halaman yang tersimpan if (kunci[0]) cached = await c.match(kunci[0]); } return cached || jaringan; }) ); });
