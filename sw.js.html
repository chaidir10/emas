// sw.js — simpan di folder yang sama dengan index.html const CACHE = 'kadar-emas-v4'; self.addEventListener('install', (e) => { e.waitUntil( caches.open(CACHE).then((c) => c.add(new Request('./', { cache: 'reload' }))) ); self.skipWaiting(); }); self.addEventListener('activate', (e) => { e.waitUntil( caches.keys() .then((keys) => Promise.all(keys.filter((k) => k !== CACHE).map((k) => caches.delete(k)))) .then(() => self.clients.claim()) ); }); // Tampilkan dari cache dulu, lalu perbarui cache di latar belakang bila online self.addEventListener('fetch', (e) => { const req = e.request; if (req.method !== 'GET' || new URL(req.url).origin !== location.origin) return; e.respondWith((async () => { const cache = await caches.open(CACHE); let cached = await cache.match(req, { ignoreSearch: true }); if (!cached && req.mode === 'navigate') cached = await cache.match('./'); const dariJaringan = fetch(req) .then((res) => { if (res.ok && !res.redirected) cache.put(req, res.clone()); return res; }) .catch(() => cached || Response.error()); if (cached) { e.waitUntil(dariJaringan); return cached; } return dariJaringan; })()); });
